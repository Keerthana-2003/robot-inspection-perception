cmake_minimum_required(VERSION 3.16)
project(perception_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(perception INTERFACE)
target_include_directories(perception INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

enable_testing()
add_executable(test_geometry tests/test_geometry.cpp)
target_link_libraries(test_geometry PRIVATE perception)
target_compile_options(test_geometry PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME geometry COMMAND test_geometry)
